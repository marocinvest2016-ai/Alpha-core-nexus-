import streamlit as st
import json
import pandas as pd

def load_properties(path="properties.json"):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

properties = load_properties()
df = pd.DataFrame(properties)

st.set_page_config(page_title="Agence - Kelaa Sraghna", layout="wide")
st.title("Agence Immobilière — Kelaa Sraghna")

# فلتر
selected_type = st.sidebar.selectbox("Type de bien", ["Tous"] + sorted(df["type"].unique().tolist()))
max_price = st.sidebar.number_input("Prix maximum (MAD)", value=int(df["price"].max()))

filtered = df.copy()
if selected_type!= "Tous":
    filtered = filtered[filtered["type"] == selected_type]
filtered = filtered[filtered["price"] <= max_price]

for _, row in filtered.iterrows():
    cols = st.columns([1, 2])
    with cols[0]:
        st.image(row["image_url"]) # تم إصلاح الخطأ هنا
    with cols[1]:
        st.subheader(row["title"])
        st.write(f"Prix : **{row['price']:,} MAD**")
